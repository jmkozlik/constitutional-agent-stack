"""Contract properties of the affected_symbols surface itself.

These are not done_when bullets.  They cover the [authored] lines that
describe how compute_affected sources its inputs, the declared category
vocabulary, and the seeded rendering callable - the parts of the contract
the guarantees lean on without restating.
"""

from __future__ import annotations

import io
import subprocess
from pathlib import Path

import pytest

from conftest import CATEGORIES, analyze, build_repo, row_for

MOD_OLD = """\
def target(a):
    return a
"""

MOD_NEW = """\
def target(a, b):
    return a + b
"""

OTHER = """\
from mod import target


def caller():
    return target(1)
"""


def _shape(report):
    return sorted(
        (row.path, row.category, bool(row.error)) for row in report.rows
    )


@pytest.fixture
def simple(two_rev_repo):
    return two_rev_repo(
        {"mod.py": MOD_OLD, "other.py": OTHER},
        {"mod.py": MOD_NEW, "other.py": OTHER},
    )


# --------------------------------------------------------------------------
# report and row shape
# --------------------------------------------------------------------------


def test_report_names_the_revisions_it_was_asked_about(simple):
    repo, old, new = simple
    report = analyze(repo, old, new)
    assert report.old == old
    assert report.new == new


def test_rows_carry_the_declared_fields(simple):
    repo, old, new = simple
    report = analyze(repo, old, new)
    assert isinstance(report.rows, list)
    assert report.rows, "a changed signature must produce at least one row"
    for row in report.rows:
        assert isinstance(row.path, str) and row.path
        assert isinstance(row.detail, str)
        assert isinstance(row.callers, list)
        assert isinstance(row.error, bool)


def test_categories_stay_inside_the_declared_vocabulary(simple):
    repo, old, new = simple
    report = analyze(repo, old, new)
    unexpected = {row.category for row in report.rows} - CATEGORIES
    assert unexpected == set(), f"undeclared categories: {unexpected}"


def test_rows_expose_callers_unknowable(simple):
    """The flag is a declared Row attribute, so it must be observable.

    Without it there is no way to tell an honest "no unresolved sites were
    detected" from a caller list that simply never admits what it missed,
    which is the distinction three of the guarantees rest on.
    """
    repo, old, new = simple
    report = analyze(repo, old, new)
    missing = [row.path for row in report.rows
               if not hasattr(row, "callers_unknowable")]
    assert missing == [], (
        f"rows carry no callers_unknowable attribute: {missing}"
    )


def test_one_row_per_affected_symbol_path(simple):
    repo, old, new = simple
    report = analyze(repo, old, new)
    seen = [row.path for row in report.rows]
    assert len(seen) == len(set(seen)), f"duplicate paths in {seen}"


# --------------------------------------------------------------------------
# category vocabulary coverage beyond the guarantees
# --------------------------------------------------------------------------


def test_a_new_symbol_is_categorised_added(two_rev_repo):
    repo, old, new = two_rev_repo(
        {"mod.py": MOD_OLD},
        {"mod.py": MOD_OLD + "\n\ndef fresh():\n    return 2\n"},
    )
    report = analyze(repo, old, new)
    assert row_for(report, "mod.fresh").category == "added"


def test_a_decorator_swap_is_categorised_decorators(two_rev_repo):
    before = """\
def deco(f):
    return f


def deco2(f):
    return f


@deco
def target():
    return 1
"""
    after = before.replace("@deco\n", "@deco2\n")
    repo, old, new = two_rev_repo({"mod.py": before}, {"mod.py": after})
    report = analyze(repo, old, new)
    assert row_for(report, "mod.target").category == "decorators"


# --------------------------------------------------------------------------
# how compute_affected sources its inputs
# --------------------------------------------------------------------------


def test_repo_accepts_str_and_path(simple):
    repo, old, new = simple
    assert _shape(analyze(str(repo), old, new)) == _shape(analyze(Path(repo), old, new))


def test_any_rev_parseable_revision_is_accepted(simple):
    repo, old, new = simple
    by_sha = analyze(repo, old, new)
    by_relative = analyze(repo, "HEAD~1", "HEAD")
    assert _shape(by_relative) == _shape(by_sha)


def test_no_index_artifact_is_written_into_the_repo(simple):
    repo, old, new = simple
    before = {
        p.relative_to(repo)
        for p in repo.rglob("*")
        if ".git" not in p.parts
    }
    analyze(repo, old, new)
    after = {
        p.relative_to(repo)
        for p in repo.rglob("*")
        if ".git" not in p.parts
    }
    assert after == before, f"compute_affected left {after - before} behind"


def test_uncommitted_worktree_state_is_never_consulted(simple):
    repo, old, new = simple
    clean = _shape(analyze(repo, old, new))

    garbage = "this file does not parse at all ][\n"
    (repo / "mod.py").write_text(garbage)
    (repo / "intruder.py").write_text("def sneaky():\n    return 1\n")

    dirty = _shape(analyze(repo, old, new))
    assert dirty == clean, (
        "uncommitted worktree state changed the report; the contract says "
        "both indexes are built from git objects at the named revisions"
    )
    assert (repo / "mod.py").read_text() == garbage, (
        "neither rev may be checked out, so the worktree must be untouched"
    )


def test_neither_revision_is_checked_out(simple):
    repo, old, new = simple
    head_before = subprocess.run(
        ["git", "rev-parse", "HEAD"], cwd=str(repo),
        capture_output=True, text=True, check=True,
    ).stdout.strip()
    analyze(repo, old, new)
    head_after = subprocess.run(
        ["git", "rev-parse", "HEAD"], cwd=str(repo),
        capture_output=True, text=True, check=True,
    ).stdout.strip()
    assert head_after == head_before


def test_an_older_pair_of_revisions_is_analysed_on_its_own_terms(tmp_path):
    """Three commits deep: the report must describe old..new, not HEAD."""
    root = tmp_path / "deep"
    first, second, third = build_repo(
        root,
        [
            {"mod.py": MOD_OLD, "other.py": OTHER},
            {"mod.py": MOD_NEW, "other.py": OTHER},
            {"mod.py": MOD_NEW + "\n\ndef later():\n    return 0\n",
             "other.py": OTHER},
        ],
    )
    report = analyze(root, first, second)
    assert row_for(report, "mod.target").category == "signature"
    assert [row.path for row in report.rows if row.path == "mod.later"] == [], (
        "a symbol introduced after new_rev must not appear in the report"
    )


# --------------------------------------------------------------------------
# the seeded rendering callable
# --------------------------------------------------------------------------


def test_render_report_writes_the_report_to_the_given_stream(simple):
    from analyzer.affected import render_report

    repo, old, new = simple
    report = analyze(repo, old, new)
    sink = io.StringIO()
    assert render_report(report, out=sink) is None
    written = sink.getvalue()
    assert written.strip(), "render_report produced no output"
    assert "mod.target" in written, (
        f"the affected symbol should appear in the rendering; got {written!r}"
    )
